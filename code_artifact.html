Google Sheets (GAS) 後端串接設定

將您部署完成的 Web App 網址貼於上方，即可直接將 1B (STEAM)、1C/2A/3B (CL) 與 F4/F6 (ICT) 的課堂資料自動存入 Google 試算表中。

協作平台嵌入建議：若將此頁面嵌入 Google Sites 或學校協作平台，建議將嵌入區塊的高度設定為 850px 以上，即可獲得完整視圖並避免出現雙重滾動條 (Double Scrollbars)。

3 分鐘建置 Google Sheets 自動紀錄檔步驟：

  1. 開啟一個全新的 Google 試算表 (Google Sheets)。
  2. 點擊上方選單 「擴充功能 (Extensions)」 > 「Apps Script」。
  3. 將下方區塊的程式碼完全覆蓋貼入編輯器中，點選儲存。
  4. 點擊右上角 「部署 (Deploy)」 > 「新建部署 (New deployment)」。
  5. 種類選擇 「Web 應用程式 (Web app)」，存取權限設為 「所有人 (Anyone)」（必須設定為所有人）。
  6. 複製部署完成後的 Web App URL 貼回本系統上方點選「儲存設定」即可！
Google Apps Script (GAS) 完整程式碼： 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // 若試算表為空，自動建立表頭標題
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "紀錄時間", "課堂日期", "班別", "科目", "課題名稱", 
        "課堂細節摘要", "是否有功課", "功課名稱", "繳交日期", 
        "專屬模組/範疇", "教具/軟體/工具", "技能/指標/實作率", "專屬備註/學習難點"
      ]);
    }

    var data = JSON.parse(e.postData.contents);
    
    // 根據班別提取專屬欄位
    var moduleVal = data.steamModule || data.clModule || data.ictModule || "";
    var toolVal = data.steamEquipment || data.clSoftware || data.ictExamFocus || "";
    var metricVal = data.steamSkill || data.clHandsOn || data.ictTool || "";
    var notesVal = data.steamGroupNotes || data.clNotes || data.ictNotes || "";

    sheet.appendRow([
      new Date(),
      data.date || "",
      data.className || "",
      data.subject || "",
      data.topic || "",
      data.details || "",
      data.hasHomework ? "是" : "否",
      data.hwTitle || "",
      data.hwDueDate || "",
      moduleVal,
      toolVal,
      metricVal,
      notesVal
    ]);

    return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
